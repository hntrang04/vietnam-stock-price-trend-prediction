#Tạo dữ liệu với loockback
LOOK_BACK = 30
def create_dataset(data, look_back=30):
    X, y = [], []
    for i in range(look_back, len(data)):
        X.append(data[i-look_back:i])
        y.append(data[i])
    return np.array(X), np.array(y)

X, y = create_dataset(scaled_close, LOOK_BACK)

# Dự đoán
def inverse_and_metrics(X, y_true, label):
    y_pred = svr.predict(X)
    y_true_inv = scaler.inverse_transform(y_true.reshape(-1,1)).ravel()
    y_pred_inv = scaler.inverse_transform(y_pred.reshape(-1,1)).ravel()

    mae = mean_absolute_error(y_true_inv, y_pred_inv)
    mape = np.mean(np.abs((y_true_inv - y_pred_inv) / y_true_inv)) * 100
    rmse = np.sqrt(mean_squared_error(y_true_inv, y_pred_inv))

    print(f"{label} - MAE: {mae:.2f}, MAPE: {mape:.2f}%, RMSE: {rmse:.2f}")
    return y_true_inv, y_pred_inv, mape

val_true, val_pred, mape_val = inverse_and_metrics(X_val, y_val, "Validation")
test_true, test_pred, mape_test = inverse_and_metrics(X_test, y_test, "Test")
