# Hàm create_dataset với window sliding    
def create_dataset(dataset, time_step):
    target_index=0               # target_index = 0 tương ứng với 'Close'
    dataX, dataY = [], []
    dataset = np.array(dataset)  # Đảm bảo rằng dataset là numpy array
    for i in range(len(dataset) - time_step):
        dataX.append(dataset[i:(i + time_step)])           # Tạo giá trị đầu vào
        dataY.append(dataset[i + time_step][target_index]) # Tạo giá trị đầu ra
    return np.array(dataX), np.array(dataY)

def inverse_close_only(pred, scaler, close_index=0, num_features=num_feature):
    temp = np.zeros((pred.shape[0], num_features))
    temp[:, close_index] = pred[:, 0]
    inv = scaler.inverse_transform(temp)
    return inv[:, close_index]

# Hàm dự đoán n ngày tiếp theo cho dữ liệu có nhiều features, chỉ dự đoán 'Close'
def predict_future(model, last_sequence, n_days, scaler, time_step, num_feature):
    future_preds = []
    current_input = last_sequence.copy()  # shape: (time_step, num_feature)

    for _ in range(n_days):
        # Dự đoán giá 'Close' tiếp theo (vị trí index 0)
        pred = model.predict(current_input.reshape(1, time_step, num_feature), verbose=1)[0][0]
        future_preds.append(pred)

        # Tạo vector có num_feature chiều, chỉ gán pred vào 'Close', còn lại giữ 0
        next_input = np.zeros((num_feature,))
        next_input[0] = pred

        # Cập nhật input sequence
        current_input = np.vstack((current_input[1:], next_input))

    # Chuẩn bị mảng cho inverse_transform (12 cột)
    future_array_scaled = np.zeros((n_days, num_feature))
    future_array_scaled[:, 0] = future_preds

    # Trả về cột 'Close' đã inverse
    return scaler.inverse_transform(future_array_scaled)[:, 0]
