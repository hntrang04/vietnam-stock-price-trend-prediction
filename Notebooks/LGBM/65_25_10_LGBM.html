def preprocess(df):
    df = df.copy()
    df.ffill(inplace=True)
    if 'price' not in df.columns:
        price_col = df.columns[0]
    else:
        price_col = 'price'
    for lag in [1, 2, 3, 5, 10]:
        df[f'lag_{lag}'] = df[price_col].shift(lag)
    df.dropna(inplace=True)
    return df, price_col

processed_result = preprocess(df)
df_proc = processed_result[0]
price_col = processed_result[1]
